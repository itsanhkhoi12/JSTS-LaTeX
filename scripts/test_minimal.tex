\subsection*{JavaScript Snippet}
\begin{minted}[frame=lines,linenos]{javascript}
// Function to compute factorial
function factorial(n) {
  if (n <= 1) return 1;
  return n * factorial(n - 1);
}
console.log(`Factorial of 5: ${factorial(5)}`);
\end{minted}
